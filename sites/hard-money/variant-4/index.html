// Set dynamic keys into authorization headers
const prismClient = axios.create({
  baseURL: 'https://api.prismlogic.io/v1/vault',
  headers: { 'Authorization': 'Bearer YOUR_PRISM_LOGIC_KEY' }
});

// Trigger matching records index
async function triggerMatch(state, loanType) {
  const response = await prismClient.get('/nodes/match', {
    params: { state, program: loanType }
  });
  return response.data;
}
