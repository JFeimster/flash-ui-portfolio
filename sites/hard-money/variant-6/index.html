// Stencil for Git direct commit of Active Vault Database
async function pushToGitRepository(vaultPayload) {
  const GITHUB_ACCESS_TOKEN = document.getElementById('setting-gh-key').value;
  const repo = document.getElementById('gh-repo-input').value;
  const branch = document.getElementById('gh-branch-input').value;
  const endpoint = `https://api.github.com/repos/${repo}/contents/database.json`;
  
  const response = await fetch(endpoint, {
    method: "PUT",
    headers: {
      "Authorization": `Bearer ${GITHUB_ACCESS_TOKEN}`,
      "Accept": "application/vnd.github.v3+json"
    },
    body: JSON.stringify({
      message: "Sync: Live Vault configuration node added.",
      content: btoa(JSON.stringify(vaultPayload)),
      branch: branch
    })
  });
  return response.json();
}

        "website": partnerDetails["Lender Website"],

        "association_stage": stage,
        "lending_cap": partnerDetails["Ltv"]
      }
    })
  });
